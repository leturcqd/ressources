\begin{enumerate}
\item[1.]
Posez et effectuez les calculs suivants : 
\begin{enumerate}
\item $12,34+9,17=$
\item $712,4+3,07=$
\item $14,92-3,5=$
\item $17,13-8,04=$
\item $4\times 7,04=$
\item $2,15\times 3,4=$
\item La division euclidienne de $127$ par $7$ (on veut le quotient et le reste).

\end{enumerate} 
\item[2.] 
Donner la valeur approchée de $12,349$ au dixième, puis au centième. 
\item[3.] Simplifier au maximum la fraction $\frac{54}{60}$.
\end{enumerate}
