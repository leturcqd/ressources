	\subsection*{Exercice 1 : Calculs (10 points)}
	\begin{enumerate}
	\item Calculer en détaillant les étapes les opérations suivantes : 
	\begin{enumerate}
	\item $5 + 3 \times 2$. 
	\item $16 \div 2 \times 4$
	\item $ 2 \times 7 + 8 \div 2$ 
	\item $(-2) + 3 + 5 - 7$
	\item $(-3) + 12 + 4 + (-9) + (-1)$ 
	\item $9- (-4) + 3$
	\item $-2 - 5$
	\item $-3 + 3$
	\end{enumerate}
	\item Réduire au même dénominateur les fractions $\frac5{6}$, $\frac23$ et $\frac{1}4$.  
	
	\item Donner l'écriture décimale de $\frac38$. 
	\end{enumerate}
